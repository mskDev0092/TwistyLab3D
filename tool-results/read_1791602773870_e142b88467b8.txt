     3→     3→     3→/**
     4→     4→     4→ * RubiksCubeCanvas
     5→     5→     5→ * ================
     6→     6→     6→ * The ONLY place where Three.js touches the DOM. This component is never
     7→     7→     7→ * server-rendered: `RubiksCubeWrapper` loads it through `next/dynamic` with
     8→     8→     8→ * `{ ssr: false }`, so `window` / `document` are guaranteed to exist.
     9→     9→     9→ *
    10→    10→    10→ * Everything (renderer, scene, camera, controls, event listeners, animation
    11→    11→    11→ * frames) is created inside a single `useEffect` and destroyed in its cleanup
    12→    12→    12→ * function, which makes hot-reloading and route changes leak-free.
    13→    13→    13→ *
    14→    14→    14→ * Interaction model
    15→    15→    15→ * -----------------
    16→    16→    16→ * - Pointer down on the BACKGROUND -> OrbitControls orbits the camera.
    17→    17→    17→ * - Pointer down on a CUBIE        -> we take over the gesture:
    18→    18→    18→ *     1. Raycast to find the cubie + the exact face that was clicked.
    19→    19→    19→ *     2. Build a fixed "drag plane" through the hit point, perpendicular to
    20→    20→    20→ *        the face normal. Every pointer position is projected onto this
    21→    21→    21→ *        plane, which turns 2D mouse movement into stable 3D movement.
    22→    22→    22→ *     3. Once the pointer moves past a small threshold, the drag direction is
    23→    23→    23→ *        snapped to the nearest principal axis lying in the face plane. The
    24→    24→    24→ *        rotation axis is `faceNormal x dragDir` - for perpendicular principal
    25→    25→    25→ *        axes the cross product is exactly the third principal axis, so no
    26→    26→    26→ *        error can creep in.
    27→    27→    27→ *     4. The 9-cubie slice around that axis is re-parented under a temporary
    28→    28→    28→ *        pivot group (see RubiksCubeLogic) and follows the pointer 1:1.
    29→    29→    29→ *     5. On release the angle snaps to the nearest multiple of 90 deg with a
    30→    30→    30→ *        short tween, then the logical grid state is committed.
    31→    31→    31→ */

    97→    97→    97→const easeOutCubic = (t: number): number => 1 - Math.pow(1 - t, 3);
    98→    98→    98→const easeInOutQuad = (t: number): number =>
    99→    99→    99→  t < 0.5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2;

   101→   101→   101→/** Push a numeric tween onto `list`; the RAF loop advances and resolves it. */
   102→   102→   102→function runTween(
   103→   103→   103→  list: TweenTask[],
   104→   104→   104→  from: number,
   105→   105→   105→  to: number,
   106→   106→   106→  duration: number,
   107→   107→   107→  ease: (t: number) => number,
   108→   108→   108→  onUpdate: (value: number) => void,
   109→   109→   109→): Promise<void> {
   110→   110→   110→  return new Promise<void>((resolve) => {
   111→   111→   111→    list.push({
   112→   112→   112→      from,
   113→   113→   113→      to,
   114→   114→   114→      duration,
   115→   115→   115→      start: performance.now(),
   116→   116→   116→      ease,
   117→   117→   117→      onUpdate,
   118→   118→   118→      resolve,
   119→   119→   119→    });
   120→   120→   120→  });
   121→   121→   121→}

   123→   123→   123→/** The two unit principal axes perpendicular to a snapped face normal. */
   124→   124→   124→function tangentAxes(normal: THREE.Vector3): [THREE.Vector3, THREE.Vector3] {
   125→   125→   125→  if (normal.x !== 0) {
   126→   126→   126→    return [new THREE.Vector3(0, 1, 0), new THREE.Vector3(0, 0, 1)];
   127→   127→   127→  }
   128→   128→   128→  if (normal.y !== 0) {
   129→   129→   129→    return [new THREE.Vector3(1, 0, 0), new THREE.Vector3(0, 0, 1)];
   130→   130→   130→  }
   131→   131→   131→  return [new THREE.Vector3(1, 0, 0), new THREE.Vector3(0, 1, 0)];
   132→   132→   132→}

   366→   366→   366→    const pickCubie = (

   369→   369→   369→      setPointerFromEvent(event);

   371→   371→   371→      if (!hit || !hit.face) return null;
   372→   372→   372→      const userData = hit.object.userData as {
   373→   373→   373→        cubie?: Cubie;
   374→   374→   374→        localNormal?: THREE.Vector3;
   375→   375→   375→      };
   376→   376→   376→      const cubie = userData.cubie;
   377→   377→   377→      if (!cubie) return null;
   378→   378→   378→
   379→   379→   379→      // Stickers know their exact outward normal; for hits on the black
   380→   380→   380→      // plastic body we derive it from the triangle normal instead.
   381→   381→   381→      const worldNormal = snapToPrincipalAxis(
   382→   382→   382→        userData.localNormal
   383→   383→   383→          ? userData.localNormal
   384→   384→   384→              .clone()

   386→   386→   386→          : hit.face.normal.clone().transformDirection(hit.object.matrixWorld),
   387→   387→   387→      );
   388→   388→   388→      return { cubie, point: hit.point.clone(), worldNormal };
   389→   389→   389→    };
   390→   390→   390→
   391→   391→   391→    const onPointerDown = (event: PointerEvent): void => {
   392→   392→   392→      if (event.target !== canvas) return; // HUD clicks never reach the cube
   393→   393→   393→      controls.autoRotate = false;
   394→   394→   394→      // Never start a new slice while the previous snap tween is settling,
   395→   395→   395→      // otherwise two pivots would fight over the same cubies.
   396→   396→   396→      if (busyRef.current || tweensRef.current.length > 0) return;
   397→   397→   397→
   398→   398→   398→      const hit = pickCubie(event);
   399→   399→   399→      if (!hit) return; // background -> OrbitControls keeps the gesture
   400→   400→   400→
   401→   401→   401→      controls.enabled = false; // freeze camera orbit for this gesture
   402→   402→   402→      canvas.style.cursor = "grabbing";
   403→   403→   403→      dragRef.current = {
   404→   404→   404→        cubie: hit.cubie,
   405→   405→   405→        startPoint: hit.point,
   406→   406→   406→        plane: new THREE.Plane().setFromNormalAndCoplanarPoint(
   407→   407→   407→          hit.worldNormal,
   408→   408→   408→          hit.point,
   409→   409→   409→        ),
   410→   410→   410→        worldNormal: hit.worldNormal,
   411→   411→   411→        locked: false,
   412→   412→   412→        dragDir: null,
   413→   413→   413→        angle: 0,
   414→   414→   414→      };
   415→   415→   415→    };
   416→   416→   416→
   417→   417→   417→    const onPointerMove = (event: PointerEvent): void => {
   418→   418→   418→      const drag = dragRef.current;
   419→   419→   419→      if (!drag) return;
   420→   420→   420→      setPointerFromEvent(event);
   421→   421→   421→
   422→   422→   422→      const current = new THREE.Vector3();
   423→   423→   423→      if (raycaster.ray.intersectPlane(drag.plane, current) === null) return;
   424→   424→   424→      const dragVector = current.sub(drag.startPoint);
   425→   425→   425→
   426→   426→   426→      if (!drag.locked) {
   427→   427→   427→        if (dragVector.length() < LOCK_THRESHOLD) return;
   428→   428→   428→
   429→   429→   429→        // Choose the in-plane principal axis that best matches the pointer
   430→   430→   430→        // movement, signed so it points the way the pointer travels.
   431→   431→   431→        const [t1, t2] = tangentAxes(drag.worldNormal);
   432→   432→   432→        const t = Math.abs(dragVector.dot(t1)) >= Math.abs(dragVector.dot(t2)) ? t1 : t2;
   433→   433→   433→        const dragDir = t.clone().multiplyScalar(Math.sign(dragVector.dot(t)) || 1);
   434→   434→   434→
   435→   435→   435→        // axis = faceNormal x dragDir. Both are perpendicular principal axes,
   436→   436→   436→        // so the result is EXACTLY +/- the third principal axis.
   437→   437→   437→        const axisVector = new THREE.Vector3().crossVectors(drag.worldNormal, dragDir);
   438→   438→   438→        const { axis, sign } = dominantAxis(axisVector);
   439→   439→   439→        const layer = Math.round(drag.cubie.grid[axis]) as -1 | 0 | 1;
   440→   440→   440→
   441→   441→   441→        logic.grabSlice(axis, layer, sign);
   442→   442→   442→        drag.locked = true;
   443→   443→   443→        drag.dragDir = dragDir;
   444→   444→   444→      }
   445→   445→   445→
   446→   446→   446→      if (!drag.dragDir) return;
   447→   447→   447→      const angle = THREE.MathUtils.clamp(
   448→   448→   448→        dragVector.dot(drag.dragDir) * (HALF_PI / DRAG_QUARTER_DISTANCE),
   449→   449→   449→        -Math.PI,
   450→   450→   450→        Math.PI,
   451→   451→   451→      );
   452→   452→   452→      drag.angle = angle;
   453→   453→   453→      logic.rotateSliceTo(angle);
   454→   454→   454→    };
   455→   455→   455→
   456→   456→   456→    const onPointerUp = (): void => {
   457→   457→   457→      const drag = dragRef.current;
   458→   458→   458→      if (!drag) return;
   459→   459→   459→      dragRef.current = null;
   460→   460→   460→      controls.enabled = true;
   461→   461→   461→      canvas.style.cursor = "grab";
   462→   462→   462→      if (!drag.locked || !drag.dragDir) return; // plain click, no twist
   463→   463→   463→
   464→   464→   464→      // Snap the slice to the nearest quarter-turn and commit it.
   465→   465→   465→      const snappedTurns = Math.round(drag.angle / HALF_PI);
   466→   466→   466→      const target = snappedTurns * HALF_PI;
   467→   467→   467→      void runTween(
   468→   468→   468→        tweensRef.current,
   469→   469→   469→        drag.angle,
   470→   470→   470→        target,
   471→   471→   471→        SNAP_DURATION_MS,
   472→   472→   472→        easeOutCubic,
   473→   473→   473→        (value) => logic.rotateSliceTo(value),
   474→   474→   474→      ).then(() => {
   475→   475→   475→        if (disposedRef.current) return;
   476→   476→   476→        logic.releaseSlice(snappedTurns);
   477→   477→   477→        if (snappedTurns === 0) return;
   478→   478→   478→
   479→   479→   479→        const nextMoves = movesRef.current + Math.abs(snappedTurns);
   480→   480→   480→        movesRef.current = nextMoves;
   481→   481→   481→        setMoves(nextMoves);
   482→   482→   482→        setFacelets(logic.getFaceletString());
   483→   483→   483→
   484→   484→   484→        const nowSolved = logic.isSolved();
   485→   485→   485→        setCubeSolved(nowSolved);
   486→   486→   486→        if (nowSolved && scrambledRef.current) {
   487→   487→   487→          scrambledRef.current = false;
   488→   488→   488→          toast.success(`Cube solved in ${nextMoves} moves!`, {
   489→   489→   489→            description: "Scramble again to keep the challenge going.",
   490→   490→   490→          });
   491→   491→   491→        }
   492→   492→   492→      });
   493→   493→   493→    };
   494→   494→   494→
   495→   495→   495→    // Capture phase on the container: runs BEFORE OrbitControls' own
   496→   496→   496→    // pointerdown handler on the canvas, so we can disable orbiting first.
   497→   497→   497→    container.addEventListener("pointerdown", onPointerDown, true);
   498→   498→   498→    window.addEventListener("pointermove", onPointerMove);
   499→   499→   499→    window.addEventListener("pointerup", onPointerUp);

   503→   503→   503→    const resize = (): void => {

   505→   505→   505→      const height = Math.max(1, container.clientHeight);
   506→   506→   506→      camera.aspect = width / height;
   507→   507→   507→      // Portrait / narrow screens: zoom out proportionally so the whole cube
   508→   508→   508→      // still fits horizontally (OrbitControls dollies the POSITION, so this
   509→   509→   509→      // zoom factor never fights with the user's scroll / pinch zoom).
   510→   510→   510→      camera.zoom = Math.min(1, 1.36 * camera.aspect);
   511→   511→   511→      camera.updateProjectionMatrix();
   512→   512→   512→      renderer.setSize(width, height);
   513→   513→   513→    };
   514→   514→   514→    const observer = new ResizeObserver(resize);
   515→   515→   515→    observer.observe(container);
   516→   516→   516→
   517→   517→   517→    const stepTweens = (now: number): void => {
   518→   518→   518→      const tasks = tweensRef.current;
   519→   519→   519→      for (let i = tasks.length - 1; i >= 0; i -= 1) {
   520→   520→   520→        const task = tasks[i];
   521→   521→   521→        const progress = Math.min(
   522→   522→   522→          1,
   523→   523→   523→          (now - task.start) / Math.max(1, task.duration),
   524→   524→   524→        );
   525→   525→   525→        task.onUpdate(task.from + (task.to - task.from) * task.ease(progress));
   526→   526→   526→        if (progress >= 1) {
   527→   527→   527→          tasks.splice(i, 1);
   528→   528→   528→          task.resolve();
   529→   529→   529→        }
   530→   530→   530→      }
   531→   531→   531→    };
   532→   532→   532→
   533→   533→   533→    let frameId = 0;
   534→   534→   534→    const tick = (now: number): void => {

   537→   537→   537→      controls.update();

   539→   539→   539→    };

   566→   566→   566→      dragRef.current = null;

   568→   568→   568→  }, []);
